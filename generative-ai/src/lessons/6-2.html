Three steps

  1. Split the documents into chunks: pieces small enough to be about one thing
  2. Index them: work out every chunk's embedding once, and save them
  3. Search: embed the question, compare it with every chunk, and return the closest few

Split the notes

Brightside's notes are Markdown files with headings, so notes.py makes one chunk per heading section:

module-6/notes.py

"""Load Brightside's notes and split them into chunks: one chunk per heading section."""
from pathlib import Path

NOTES_DIR = Path(__file__).resolve().parent.parent / "data" / "brightside-notes"


def load_chunks():
    chunks = []
    for path in sorted(NOTES_DIR.glob("*.md")):
        title = ""
        section = []
        for line in path.read_text(encoding="utf-8").splitlines():
            if line.startswith("#"):
                if section:
                    chunks.append({"source": path.name, "title": title, "text": "\n".join(section).strip()})
                title = line.lstrip("# ").strip()
                section = [line]
            else:
                section.append(line)
        if section:
            chunks.append({"source": path.name, "title": title, "text": "\n".join(section).strip()})
    return [c for c in chunks if len(c["text"].split()) > 5]


if __name__ == "__main__":
    for chunk in load_chunks():
        print(f"{chunk['source']:20} {chunk['title']:35} {len(chunk['text'].split())} words")

Each chunk remembers its source file and heading, so answers can say where they came from. The last line drops chunks that are only a heading.

Index and search

module-6/search.py

"""Search Brightside's notes by meaning, not by matching words."""
import json
import math
from pathlib import Path

import ollama

from notes import load_chunks

EMBED_MODEL = "nomic-embed-text"
INDEX_FILE = Path(__file__).resolve().parent / "index.json"


def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    return dot / (math.sqrt(sum(x * x for x in a)) * math.sqrt(sum(y * y for y in b)))


def build_index():
    chunks = load_chunks()
    # nomic-embed-text works best with these prefixes
    vectors = ollama.embed(model=EMBED_MODEL,
                           input=["search_document: " + c["text"] for c in chunks])["embeddings"]
    for chunk, vector in zip(chunks, vectors):
        chunk["vector"] = vector
    INDEX_FILE.write_text(json.dumps(chunks), encoding="utf-8")
    return chunks


def load_index():
    if INDEX_FILE.exists():
        return json.loads(INDEX_FILE.read_text(encoding="utf-8"))
    return build_index()


def search(question, k=3):
    chunks = load_index()
    q = ollama.embed(model=EMBED_MODEL, input="search_query: " + question)["embeddings"][0]
    scored = sorted(chunks, key=lambda c: cosine(q, c["vector"]), reverse=True)
    return [(round(cosine(q, c["vector"]), 3), c) for c in scored[:k]]


if __name__ == "__main__":
    build_index()
    while True:
        question = input("\nSearch (or 'quit'): ")
        if question.strip().lower() == "quit":
            break
        for score, chunk in search(question):
            print(f"{score}  {chunk['source']} > {chunk['title']}")
  • build_index embeds every chunk in one call and saves the chunks and their embeddings to index.json, so you don't redo it for every question. Run it again whenever the notes change
  • search_document: and search_query: are prefixes this embedding model was trained with: they tell it which side is which, and improve results
  • search scores every chunk and returns the top k

All three found the right chunk first, though the refunds note never says "money back", the prices note says "brothers and sisters" not "siblings", and nothing says "HSC biology" in a heading.

Comparing with every chunk is fine for a few hundred chunks. For millions, apps use a vector database (like pgvector or Chroma) that finds the closest embeddings quickly.
