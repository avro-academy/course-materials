Summaries

SELECT staff_id, count(*) AS orders
FROM orders
WHERE payment = 'Card'      -- rows, before grouping
GROUP BY staff_id
HAVING count(*) > 200       -- groups, after
ORDER BY orders DESC;

count(*) · count(col) skips NULLs · sum avg min max · round(x, 2). Every SELECT column is in GROUP BY or inside an aggregate.

Joins

FROM menu_items m
JOIN categories c ON c.id = m.category_id;

SELECT s.first_name, count(o.id)
FROM staff s
LEFT JOIN orders o ON o.staff_id = s.id
GROUP BY s.first_name;
-- missing: LEFT JOIN … WHERE o.id IS NULL

Transactions

BEGIN;
UPDATE menu_items SET price = price + 0.50 WHERE category_id = 1;
SELECT name, price FROM menu_items WHERE category_id = 1;
ROLLBACK;     -- or COMMIT;

"current transaction is aborted" → ROLLBACK; and start again.

Design tables

CREATE TABLE shifts (
  id        integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  staff_id  integer NOT NULL REFERENCES staff (id),
  starts_at timestamp NOT NULL,
  ends_at   timestamp NOT NULL,
  notes     text,
  CONSTRAINT ends_after_start CHECK (ends_at > starts_at)
);
ALTER TABLE menu_items ADD COLUMN contains_nuts boolean NOT NULL DEFAULT false;
ALTER TABLE staff RENAME COLUMN a TO b;   DROP TABLE suppliers;

Types: integer · numeric(5,2) for money · text · boolean · date · timestamp. Rules: NOT NULL UNIQUE CHECK DEFAULT REFERENCES … ON DELETE CASCADE
